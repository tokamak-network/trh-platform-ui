Option B 컨트랙트 설계 스케치 (Phase 3)
EntryPointAutoRefiller.sol
contract EntryPointAutoRefiller {
    IEntryPoint    public entryPoint;       // predeploy 0x4200...0063
    IMultiTokenPaymaster public paymaster;  // predeploy 0x4200...0067
    ISwapRouter    public swapRouter;       // Uniswap V3 SwapRouter
    address        public wton;             // 0x4200...0006
    address        public feeToken;         // USDT / USDC / WETH

    uint256 public refillThreshold  = 0.5 ether;  // TON
    uint256 public swapMinFeeAmount = 10e6;        // 10 USDT (skip if less)

    /// @notice Permissionless — anyone can trigger
    function checkAndRefill() external {
        if (entryPoint.balanceOf(address(paymaster)) >= refillThreshold) return;

        // 1. Withdraw collected fees from paymaster
        uint256 feeAmt = paymaster.collectedFees(feeToken);
        if (feeAmt < swapMinFeeAmount) return;
        paymaster.withdrawCollectedFees(feeToken, address(this), feeAmt);

        // 2. Swap feeToken → WTON via Uniswap V3
        IERC20(feeToken).approve(address(swapRouter), feeAmt);
        uint256 wtonOut = swapRouter.exactInputSingle(ISwapRouter.ExactInputSingleParams({
            tokenIn: feeToken, tokenOut: wton,
            fee: 3000, recipient: address(this),
            amountIn: feeAmt, amountOutMinimum: 0, sqrtPriceLimitX96: 0
        }));

        // 3. Unwrap WTON → TON
        IWTON(wton).withdraw(wtonOut);

        // 4. Deposit TON to EntryPoint
        entryPoint.depositTo{value: wtonOut}(address(paymaster));
    }

    receive() external payable {}
}
⚠️ MultiTokenPaymaster.withdrawCollectedFees의 호출권 → Refiller 컨트랙트를 owner로 설정하거나 별도 withdrawer role 추가 필요 (현재 MultiTokenPaymaster는 owner만 인출 가능).

func (t *ThanosStack) startEntryPointRefillMonitor(ctx context.Context) {
    go func() {
        ticker := time.NewTicker(5 * time.Minute)
        defer ticker.Stop()
        for {
            select {
            case <-ctx.Done():
                return
            case <-ticker.C:
                if err := t.checkAndRefillEntryPoint(ctx); err != nil {
                    t.logger.Warnf("EntryPoint refill check failed: %v", err)
                }
            }
        }
    }()
}

요약:
• 오라클(UniswapV3TwapOracle)은 완전 자동 — L2 배포 후 별도 액션 없이 영구 동작.
• EntryPoint TON 잔고는 오라클과 별개 문제. Option A (backend poller)로 자동화 가능, 어드민 지갑에 TON이 충분하면 됨.
• 장기적으로는 Option B (self-funded cycle: USDT 수수료 → Uniswap 스왑 → TON → depositTo)가 완전 무인화 가능. MultiTokenPaymaster에 withdrawer role 추가가 선행 필요.
